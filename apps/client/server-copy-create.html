// apps/client/api/server-copy.js export default async function handler(req, res) { // শুধুমাত্র GET মেথড অনুমতি দাও if (req.method !== 'GET') { return res.status(405).json({ message: 'Method Not Allowed' }); } const { nid, dob } = req.query; const apiKey = '2a3b267ca79e6abc23fb675a0e381656'; // লক্ষ্য করো: এখানে সরাসরি servercopy.work এর URL ব্যবহার করা হচ্ছে const targetUrl = `https://servercopy.work/sv.php?key=${apiKey}&nid=${nid}&dob=${dob}`; try { // Vercel Functions এর ডিফল্ট টাইমআউট ৬ সেকেন্ড, তাই এক্সটেন্ড করা হচ্ছে const response = await fetch(targetUrl, { headers: { 'Accept': 'application/json', 'User-Agent': 'NagarikSeba-Vercel-Bot' }, // টাইমআউট বাড়ানো হয়েছে (ms) signal: AbortSignal.timeout(30000) }); if (!response.ok) { throw new Error(`Server Error: ${response.status}`); } const data = await response.json(); // CORS হেডার যোগ করা যাতে ব্রাউজার ডাটা পড়তে পারে res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); res.status(200).json(data); } catch (error) { console.error("API Fetch Error:", error); res.status(500).json({ success: false, message: "সার্ভার সংযোগ ব্যর্থ হয়েছে। আবার চেষ্টা করুন।", error: error.message }); } }
